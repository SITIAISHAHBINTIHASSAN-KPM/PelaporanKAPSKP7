/**
 * SISTEM ANALISIS PEPERIKSAAN KELAS AGAMA PETANG (KAP)
 * SK (FELDA) PALONG 7
 * Spreadsheet ID: 1NEjOwcydF3Z1RqSVfCcxDXjApT9RCNkh2DoKdJ9hQmo
 */

function doGet(e) {
  var ss = SpreadsheetApp.openById('1NEjOwcydF3Z1RqSVfCcxDXjApT9RCNkh2DoKdJ9hQmo');
  var sheetName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : 'MarkahKAP';
  var sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(['StudentID', 'Nama', 'Kelas', 'Pentaksiran', 'Subjek', 'Markah', 'Gred', 'Tafsiran', 'Timestamp']);
  }
  
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.openById('1NEjOwcydF3Z1RqSVfCcxDXjApT9RCNkh2DoKdJ9hQmo');
  var sheetName = 'MarkahKAP';
  var sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(['StudentID', 'Nama', 'Kelas', 'Pentaksiran', 'Subjek', 'Markah', 'Gred', 'Tafsiran', 'Timestamp']);
  }
  
  try {
    var payload = JSON.parse(e.postData.contents);
    var rows = payload.records || [];
    
    for (var i = 0; i < rows.length; i++) {
      var r = rows[i];
      sheet.appendRow([
        r.studentId,
        r.nama,
        r.kelas,
        r.pentaksiran,
        r.subjek,
        r.markah,
        r.gred,
        r.tafsiran,
        new Date()
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Rekod markah berjaya disimpan' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
